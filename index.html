-- ========================================================= -- ZIVOZONE DATABASE v1 -- Profiles + Games + XP + ZIVO Ledger + Challenges -- ========================================================= create extension if not exists pgcrypto; -- ========================================================= -- 1. PLAYER PROFILES -- ========================================================= create table if not exists public.profiles ( id uuid primary key references auth.users(id) on delete cascade, username text not null check (char_length(username) between 2 and 40), age integer not null check (age between 5 and 100), avatar_url text, xp integer not null default 0 check (xp >= 0), level integer not null default 1 check (level >= 1), zivo_balance bigint not null default 0 check (zivo_balance >= 0), games_played integer not null default 0 check (games_played >= 0), wins integer not null default 0 check (wins >= 0), streak integer not null default 0 check (streak >= 0), created_at timestamptz not null default now(), updated_at timestamptz not null default now() ); -- ========================================================= -- 2. GAME RESULTS -- ========================================================= create table if not exists public.game_results ( id uuid primary key default gen_random_uuid(), player_id uuid not null references public.profiles(id) on delete cascade, game_type text not null check ( game_type in ('quiz', 'science', 'horror', 'daily', 'identity') ), score integer not null default 0 check (score >= 0), total_questions integer not null default 0 check (total_questions >= 0), xp_earned integer not null default 0 check (xp_earned >= 0), zivo_earned integer not null default 0 check (zivo_earned >= 0), duration_seconds integer check ( duration_seconds is null or duration_seconds >= 0 ), created_at timestamptz not null default now() ); -- ========================================================= -- 3. ZIVO TRANSACTIONS -- ========================================================= create table if not exists public.zivo_transactions ( id uuid primary key default gen_random_uuid(), player_id uuid not null references public.profiles(id) on delete cascade, amount bigint not null, transaction_type text not null check ( transaction_type in ( 'game_reward', 'daily_reward', 'achievement', 'purchase', 'admin_adjustment' ) ), reference_id uuid, description text, created_at timestamptz not null default now() ); -- ========================================================= -- 4. DAILY CHALLENGES -- ========================================================= create table if not exists public.daily_challenges ( id uuid primary key default gen_random_uuid(), challenge_date date not null unique, title text not null, description text not null, game_type text not null check ( game_type in ('quiz', 'science', 'horror') ), reward_xp integer not null default 30 check (reward_xp >= 0), reward_zivo integer not null default 1 check (reward_zivo >= 0), is_active boolean not null default true, created_at timestamptz not null default now() ); -- ========================================================= -- 5. ACHIEVEMENTS -- ========================================================= create table if not exists public.achievements ( id uuid primary key default gen_random_uuid(), code text not null unique, title text not null, description text not null, reward_xp integer not null default 0 check (reward_xp >= 0), reward_zivo integer not null default 0 check (reward_zivo >= 0), created_at timestamptz not null default now() ); -- ========================================================= -- 6. PLAYER ACHIEVEMENTS -- ========================================================= create table if not exists public.player_achievements ( player_id uuid not null references public.profiles(id) on delete cascade, achievement_id uuid not null references public.achievements(id) on delete cascade, earned_at timestamptz not null default now(), primary key (player_id, achievement_id) ); -- ========================================================= -- 7. IDENTITY RESULTS -- ========================================================= create table if not exists public.identity_results ( id uuid primary key default gen_random_uuid(), player_id uuid not null references public.profiles(id) on delete cascade, logic_score integer not null default 0, courage_score integer not null default 0, curiosity_score integer not null default 0, social_score integer not null default 0, personality_type text, created_at timestamptz not null default now() ); -- ========================================================= -- 8. INDEXES -- ========================================================= create index if not exists idx_game_results_player on public.game_results(player_id); create index if not exists idx_game_results_created on public.game_results(created_at desc); create index if not exists idx_zivo_transactions_player on public.zivo_transactions(player_id); create index if not exists idx_zivo_transactions_created on public.zivo_transactions(created_at desc); create index if not exists idx_identity_player on public.identity_results(player_id); -- ========================================================= -- 9. ENABLE ROW LEVEL SECURITY -- ========================================================= alter table public.profiles enable row level security; alter table public.game_results enable row level security; alter table public.zivo_transactions enable row level security; alter table public.daily_challenges enable row level security; alter table public.achievements enable row level security; alter table public.player_achievements enable row level security; alter table public.identity_results enable row level security; -- ========================================================= -- 10. PROFILE POLICIES -- ========================================================= drop policy if exists "Players can view own profile" on public.profiles; create policy "Players can view own profile" on public.profiles for select to authenticated using (auth.uid() = id); drop policy if exists "Players can create own profile" on public.profiles; create policy "Players can create own profile" on public.profiles for insert to authenticated with check (auth.uid() = id); -- لا نعطي اللاعب صلاحية مباشرة لتعديل XP أو ZIVO. -- هذه القيم ستعدل لاحقًا من خلال Backend Functions آمنة. drop policy if exists "Players can update basic profile" on public.profiles; create policy "Players can update basic profile" on public.profiles for update to authenticated using (auth.uid() = id) with check (auth.uid() = id); -- ========================================================= -- 11. GAME RESULTS POLICIES -- ========================================================= drop policy if exists "Players can view own results" on public.game_results; create policy "Players can view own results" on public.game_results for select to authenticated using (auth.uid() = player_id); -- لن نسمح بإضافة نتائج مباشرة من المتصفح في النسخة الآمنة. -- الإضافة ستتم لاحقًا عبر RPC / Edge Function. -- ========================================================= -- 12. ZIVO TRANSACTIONS -- ========================================================= drop policy if exists "Players can view own ZIVO transactions" on public.zivo_transactions; create policy "Players can view own ZIVO transactions" on public.zivo_transactions for select to authenticated using (auth.uid() = player_id); -- لا يوجد INSERT مباشر للاعب. -- ========================================================= -- 13. DAILY CHALLENGES -- ========================================================= create policy "Anyone can view active daily challenges" on public.daily_challenges for select to authenticated using (is_active = true); -- ========================================================= -- 14. ACHIEVEMENTS -- ========================================================= create policy "Anyone can view achievements" on public.achievements for select to authenticated using (true); -- ========================================================= -- 15. PLAYER ACHIEVEMENTS -- ========================================================= create policy "Players can view own achievements" on public.player_achievements for select to authenticated using (auth.uid() = player_id); -- ========================================================= -- 16. IDENTITY RESULTS -- ========================================================= create policy "Players can view own identity results" on public.identity_results for select to authenticated using (auth.uid() = player_id); -- ========================================================= -- 17. INITIAL ACHIEVEMENTS -- ========================================================= insert into public.achievements (code, title, description, reward_xp, reward_zivo) values ('FIRST_GAME', 'أول لعبة', 'أكمل أول لعبة في ZIVOZONE', 25, 1), ('FIRST_WIN', 'أول انتصار', 'حقق أول فوز', 50, 2), ('DAILY_PLAYER', 'لاعب يومي', 'أكمل تحديًا يوميًا', 30, 1), ('LEVEL_5', 'المستوى الخامس', 'وصل إلى Level 5', 100, 5) on conflict (code) do nothing; -- ========================================================= -- 18. FUNCTION: UPDATE TIMESTAMP -- ========================================================= create or replace function public.set_updated_at() returns trigger language plpgsql security invoker as $$ begin new.updated_at = now(); return new; end; $$; drop trigger if exists profiles_updated_at on public.profiles; create trigger profiles_updated_at before update on public.profiles for each row execute function public.set_updated_at(); -- ========================================================= -- DONE -- =========================================================
